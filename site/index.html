effects/
└─ your-effect/
   ├─ index.html   # 双击就能打开的 Demo
   ├─ style.css    # 参数都放在 :root 变量里
   ├─ script.js    # 可选
   ├─ meta.json    # 名字、分类、参数
   └─ README.md
